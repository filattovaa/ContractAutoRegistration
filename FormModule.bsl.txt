&НаКлиенте
Процедура ПутьКФайлуНачалоВыбора(Элемент, ДанныеВыбора, ВыборДобавлением, СтандартнаяОбработка)
	
	СтандартнаяОбработка = Ложь;
	Диалог = Новый ДиалогВыбораФайла(РежимДиалогаВыбораФайла.Открытие);
	Диалог.МножественныйВыбор = Ложь;
	Диалог.Фильтр = "Табличные документы (*.xlsx) | *.xlsx";
	Диалог.Заголовок = "Выберите файл";
	
	Если Не Диалог.Выбрать() Тогда
		Возврат;
	КонецЕсли;
	
	ПутьКФайлу = Диалог.ПолноеИмяФайла;
	АдресФайла = "";
	
	Попытка
		ПоместитьФайлВоВременноеХранилище(АдресФайла, ПутьКФайлу);
		Если ПустаяСтрока(АдресФайла) Тогда
			ВызватьИсключение "Не удалось поместить файл во временное хранилище";
		КонецЕсли;
		ЗаполнитьТаблицу(АдресФайла);
	Исключение
		Предупреждение("Ошибка загрузки файла: " + ОписаниеОшибки());
	КонецПопытки; 
	
КонецПроцедуры

КонецПроцедуры

&НаСервере
Процедура ЗаполнитьТаблицу(АдресФайла)
	
	ДанныеФайла = ПолучитьИзВременногоХранилища(АдресФайла);

    ИмяВременногоФайла = ПолучитьИмяВременногоФайла("xlsx");
	
	Попытка
        ДанныеФайла.Записать(ИмяВременногоФайла);
        ТабДок = Новый ТабличныйДокумент;
        ТабДок.Прочитать(ИмяВременногоФайла);
    Исключение
        ВызватьИсключение "Не удалось прочитать Excel: " + ОписаниеОшибки();
    КонецПопытки;

    СтрокиПоКоду = Новый Соответствие;
    Коды = Новый Массив;
    Для НомерСтроки = 2 По ТабДок.ВысотаТаблицы Цикл
        Код = СокрЛП(ТабДок.Область(НомерСтроки, 1).Текст);
		
		Если ПустаяСтрока(Код) Тогда
            Продолжить;
		КонецЕсли;
		
        Попытка
            ДатаНачала = ОбщегоНазначенияКлиентСервер.СтрокаВДату(
                СокрЛП(ТабДок.Область(НомерСтроки, 3).Текст));
            ДатаОкончания = ОбщегоНазначенияКлиентСервер.СтрокаВДату(
                СокрЛП(ТабДок.Область(НомерСтроки, 4).Текст));
        Исключение
            ВызватьИсключение "Некорректные даты в документе: " + Код + " - " + ОписаниеОшибки();
		КонецПопытки; 
			
        Если Не ЗначениеЗаполнено(ДатаНачала)
            Или Не ЗначениеЗаполнено(ДатаОкончания)
            Или ДатаНачала > ДатаОкончания Тогда
            ВызватьИсключение "Неверный период действия договора в строке " + Строка(НомерСтроки);
        КонецЕсли;

        СтрокиПоКоду.Вставить(Код, Новый Структура("ДатаНачала, ДатаОкончания", ДатаНачала, ДатаОкончания));
        Коды.Добавить(Код);
    КонецЦикла;

    ДокументыФайла.Очистить();
	
	Если Коды.Количество() = 0 Тогда
        Возврат;
    КонецЕсли;

    Запрос = Новый Запрос;
    Запрос.Текст =
    "ВЫБРАТЬ РАЗЛИЧНЫЕ
    |	Состояния.Документ КАК Документ,
    |	Состояния.Документ.Контрагент КАК Контрагент,
    |	Состояния.Документ.Код КАК Код
    |ПОМЕСТИТЬ ВТ_Документы
    |ИЗ
    |	РегистрСведений.ТекущиеСостоянияДокументов КАК Состояния
    |ГДЕ
    |	Состояния.Документ.Код В(&Коды)
    |	И Состояния.Состояние = &Состояние
    |;
    |
    |////////////////////////////////////////////////////////////////////////////////
    |ВЫБРАТЬ
    |	Документы.Документ КАК Документ,
    |	Документы.Контрагент КАК Контрагент,
    |	Документы.Код КАК Код,
    |	Задачи.Ссылка КАК ЗадачаРегистрации
    |ИЗ
    |	ВТ_Документы КАК Документы
    |		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Задача.ЗадачаИсполнителя.Предметы КАК ПредметыЗадач
    |		ПО (ПредметыЗадач.Предмет = Документы.Документ)
    |		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Задача.ЗадачаИсполнителя КАК Задачи
    |		ПО (Задачи.Ссылка = ПредметыЗадач.Ссылка)
    |ГДЕ
    |	Задачи.ТочкаМаршрута = &ТочкаМаршрута";
	
    Запрос.УстановитьПараметр("Коды", Коды);
    Запрос.УстановитьПараметр("Состояние", Перечисления.СостоянияДокументов.НаРегистрации);
    Запрос.УстановитьПараметр("ТочкаМаршрута", БизнесПроцессы.Регистрация.ТочкиМаршрута.Зарегистрировать);

    Выборка = Запрос.Выполнить().Выбрать();
    Пока Выборка.Следующий() Цикл
        Даты = СтрокиПоКоду.Получить(СокрЛП(Выборка.Код));
        Если Даты = Неопределено Тогда
            Продолжить;
        КонецЕсли;
        НоваяСтрока = ДокументыФайла.Добавить();
        НоваяСтрока.Документ = Выборка.Документ;
        НоваяСтрока.НомерДокумента = Выборка.Код;
        НоваяСтрока.Контрагент = Выборка.Контрагент;
        НоваяСтрока.ЗадачаРегистрации = Выборка.ЗадачаРегистрации;
        НоваяСтрока.ДатаНачалаДействия = Даты.ДатаНачала;
        НоваяСтрока.ДатаОкончанияДействия = Даты.ДатаОкончания;
    КонецЦикла;

КонецПроцедуры

&НаСервере
Процедура ЗарегистрироватьДокументы()
	
	Для Каждого СтрокаДокумента Из ДокументыФайла Цикл
		
		Если Не ЗначениеЗаполнено(СтрокаДокумента.Документ)
			Или Не ЗначениеЗаполнено(СтрокаДокумента.ЗадачаРегистрации) Тогда
			ВызватьИсключение "Не найден документ или задача регистрации";
		КонецЕсли;
		
		НачатьТранзакцию();

		Попытка
					
			ДокументОбъект = СтрокаДокумента.Документ.ПолучитьОбъект();
			ДокументОбъект.ДатаНачалаДействия = СтрокаДокумента.ДатаНачалаДействия;
			ДокументОбъект.ДатаОкончанияДействия = СтрокаДокумента.ДатаОкончанияДействия;
			ДокументОбъект.ПорядокПродления = Перечисления.ПорядокПродления.НеПродлевается;
			ДокументОбъект.Записать();
			
			ЗадачаОбъект = СтрокаДокумента.ЗадачаРегистрации.ПолучитьОбъект();
			ЗадачаОбъект.Исполнитель = Исполнитель;
			ЗадачаОбъект.ТекущийИсполнитель = Исполнитель;
			ЗадачаОбъект.Записать();
			ЗадачаОбъект.ВыполнитьЗадачу();
			
			Делопроизводство.ЗарегистрироватьПредмет(СтрокаДокумента.Документ, "", "");  
			
			РегистрыСведений.РезультатыВыполненияПроцессовИЗадач.ЗаписатьРезультатПоОбъекту(
				СтрокаДокумента.ЗадачаРегистрации,
				Перечисления.ВариантыВыполненияПроцессовИЗадач.Положительно);
		
			Делопроизводство.ЗаписатьСостояниеДокумента(
				СтрокаДокумента.Документ,
				ТекущаяДатаСеанса(),
				Перечисления.СостоянияДокументов.Зарегистрирован,
				Исполнитель);
		

			ЗафиксироватьТранзакцию();
			
		Исключение
			
			ОтменитьТранзакцию();
			
			ИмяСобытия = НСтр("ru = 'Авторегистрация договоров'",
					ОбщегоНазначения.КодОсновногоЯзыка());
			ТекстОшибки = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
			
			ЗаписьЖурналаРегистрации(ИмяСобытия, УровеньЖурналаРегистрации.Ошибка,,, ТекстОшибки);
			
			ВызватьИсключение ТекстОшибки;
				
		КонецПопытки;
		
		СтрокаДокумента.РегНомер = Строка(СтрокаДокумента.Документ.РегистрационныйНомер)
			+ " от " + Строка(СтрокаДокумента.Документ.ДатаРегистрации);   
			
	КонецЦикла; 
		
КонецПроцедуры

#КонецОбласти
